cmake_minimum_required(VERSION 3.20)
project(CMemoryToolkit VERSION 0.1.0 LANGUAGES C)
include(CTest)
option(BUILD_TESTING "Build tests" ON)
add_library(c_memory src/version.c)
target_include_directories(c_memory PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(c_memory PUBLIC c_std_17)
if(BUILD_TESTING)
    add_executable(c_memory_test tests/version_test.c)
    target_link_libraries(c_memory_test PRIVATE c_memory)
    add_test(NAME c_memory_test COMMAND c_memory_test)
endif()
install(TARGETS c_memory)
install(DIRECTORY include/ DESTINATION include)
